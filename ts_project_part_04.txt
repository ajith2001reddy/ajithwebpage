/* ═══════════════════════════════════════════════════════════════════════════
   TETHER
═══════════════════════════════════════════════════════════════════════════ */
function Tether({ worldPosRef }) {
  const lineRef = useRef();
  const ANCHOR = useMemo(() => new THREE.Vector3(-13, 8, -4), []);
  const mid = useMemo(() => new THREE.Vector3(), []);
  const end = useMemo(() => new THREE.Vector3(), []);

  useFrame(() => {
    if (!lineRef.current) return;
    end.set(worldPosRef.current.x, worldPosRef.current.y, 0);
    mid.set(
      (ANCHOR.x + end.x) * 0.5,
      (ANCHOR.y + end.y) * 0.5 + 2.0,
      (ANCHOR.z + end.z) * 0.5,
    );
    lineRef.current.geometry.setFromPoints(
      new THREE.QuadraticBezierCurve3(ANCHOR, mid, end).getPoints(60)
    );
  });

  return (
    <line ref={lineRef}>
      <bufferGeometry />
      <lineBasicMaterial color="#888888" transparent opacity={0.4} />
    </line>
  );
}

/* ═══════════════════════════════════════════════════════════════════════════
   THRUSTER PARTICLES
═══════════════════════════════════════════════════════════════════════════ */
function ThrusterParticles({ active, ox, oy, oz }) {
  const ref = useRef();
  const N = 80;
  const ages = useRef(new Float32Array(N).map(() => Math.random()));
  const vels = useRef(new Float32Array(N * 3));

  const geo = useMemo(() => {
    const g = new THREE.BufferGeometry();
    g.setAttribute('position', new THREE.BufferAttribute(new Float32Array(N * 3).fill(9999), 3));
    return g;
  }, []);

  const mat = useMemo(() => new THREE.PointsMaterial({
    color: '#ff9933', size: 0.038, transparent: true, opacity: 0.9,
    sizeAttenuation: true, depthWrite: false, blending: THREE.AdditiveBlending,
  }), []);

  useFrame((_, dt) => {
    if (!ref.current) return;
    ref.current.visible = active;
    if (!active) return;
    const pos = geo.attributes.position.array;
    const v = vels.current;
    for (let i = 0; i < N; i++) {
      ages.current[i] -= dt * 3.2;
      if (ages.current[i] <= 0) {
        pos[i * 3] = ox + (Math.random() - 0.5) * 0.055;
        pos[i * 3 + 1] = oy;
        pos[i * 3 + 2] = oz + (Math.random() - 0.5) * 0.055;
        v[i * 3] = (Math.random() - 0.5) * 0.018;
        v[i * 3 + 1] = -(Math.random() * 0.065 + 0.025);
        v[i * 3 + 2] = (Math.random() - 0.5) * 0.018;
        ages.current[i] = Math.random() * 0.32 + 0.12;
      }
      pos[i * 3] += v[i * 3];
      pos[i * 3 + 1] += v[i * 3 + 1];
      pos[i * 3 + 2] += v[i * 3 + 2];
    }
    geo.attributes.position.needsUpdate = true;
  });

  return <points ref={ref} geometry={geo} material={mat} />;
}

/* ═══════════════════════════════════════════════════════════════════════════
   NASA EMU ASTRONAUT  –  full procedural geometry
═══════════════════════════════════════════════════════════════════════════ */
function NASAAstronaut({ hovered, setHovered }) {
  /* ── Refs ──────────────────────────────────────────────────────────────── */
  const rootRef = useRef();   // whole astronaut group
  const torsoRef = useRef();   // HUT + LTA
  const headRef = useRef();   // helmet assembly
  const lArmRef = useRef();   // left  upper arm pivot
  const rArmRef = useRef();   // right upper arm pivot
  const lForeRef = useRef();   // left  lower arm pivot
  const rForeRef = useRef();   // right lower arm pivot
  const lLegRef = useRef();   // left  upper leg pivot
  const rLegRef = useRef();   // right upper leg pivot
  const lShinRef = useRef();   // left  lower leg pivot
  const rShinRef = useRef();   // right lower leg pivot

  /* ── Physics ────────────────────────────────────────────────────────── */
  const worldPos = useRef(new THREE.Vector3(0, 0, 0));
  const targetPos = useRef({ x: 0, y: 0 });
  const vel = useRef({ x: 0, y: 0 });
  const actionRef = useRef('walk');
  const boostRef = useRef(0);

  /* ── React state (thought bubble only) ─────────────────────────────── */
  const [status, setStatus] = useState('👨‍🚀 On patrol...');
  const [repairing, setRepairing] = useState(false);
  const [flying, setFlying] = useState(false);

  /* ── Events ─────────────────────────────────────────────────────────── */
  useEffect(() => {
    const onAct = ({ detail }) => {
      const { action, target, status: s } = detail;
      if (action) {
        actionRef.current = action;
        setRepairing(action === 'repair');
        setFlying(action === 'fly');
      }
      if (target) targetPos.current = target;
      if (s) setStatus(s);
    };
    const onBoost = () => { boostRef.current = 1; };
    window.addEventListener('astronaut-autonomous-action', onAct);
    window.addEventListener('astro-boost', onBoost);
    return () => {
      window.removeEventListener('astronaut-autonomous-action', onAct);
      window.removeEventListener('astro-boost', onBoost);
    };
  }, []);

  /* ── Frame loop ──────────────────────────────────────────────────────── */
  useFrame((state, delta) => {
    const dt = Math.min(delta, 0.05);
    const t = state.clock.getElapsedTime();
    if (!rootRef.current) return;

    /* Physics movement */
    vel.current.x += (targetPos.current.x * 5 - worldPos.current.x) * dt * 1.9;
    vel.current.y += (targetPos.current.y * 3 - worldPos.current.y) * dt * 1.9;
    vel.current.x *= Math.pow(0.87, dt * 60);
    vel.current.y *= Math.pow(0.87, dt * 60);
    worldPos.current.x += vel.current.x;
    worldPos.current.y += vel.current.y;

    /* lerp helper */
    const lr = (cur, tgt, spd) => THREE.MathUtils.lerp(cur, tgt, dt * spd);

    /* Breathing – base layer */
    const breathe = Math.sin(t * 0.9) * 0.013;

    const action = actionRef.current;

    /* ── WALK ──────────────────────────────────────────────────────────── */
    if (action === 'walk') {
      const phase = t * 2.05;

      /* Legs swing forward/back on X axis */
      lLegRef.current.rotation.x = lr(lLegRef.current.rotation.x, Math.sin(phase) * 0.40, 14);
      rLegRef.current.rotation.x = lr(rLegRef.current.rotation.x, -Math.sin(phase) * 0.40, 14);
      /* Knee flex (lower leg swings slightly on same phase) */
      lShinRef.current.rotation.x = lr(lShinRef.current.rotation.x, Math.max(0, Math.sin(phase + 0.4)) * 0.28, 12);
      rShinRef.current.rotation.x = lr(rShinRef.current.rotation.x, Math.max(0, Math.sin(phase + 0.4 + Math.PI)) * 0.28, 12);

      /* Arms swing opposite to legs */
      lArmRef.current.rotation.x = lr(lArmRef.current.rotation.x, -Math.sin(phase) * 0.30, 12);
      rArmRef.current.rotation.x = lr(rArmRef.current.rotation.x, Math.sin(phase) * 0.30, 12);
      /* Arms hang close to body */
      lArmRef.current.rotation.z = lr(lArmRef.current.rotation.z, 0.06, 5);
      rArmRef.current.rotation.z = lr(rArmRef.current.rotation.z, -0.06, 5);
      /* Slight forearm bend always */
      lForeRef.current.rotation.x = lr(lForeRef.current.rotation.x, 0.12, 5);
      rForeRef.current.rotation.x = lr(rForeRef.current.rotation.x, 0.12, 5);

      /* Torso: hip sway + forward lean */
      torsoRef.current.rotation.z = Math.sin(phase) * 0.024;
      torsoRef.current.rotation.x = lr(torsoRef.current.rotation.x, 0.04, 3);

      /* Head stays level, very subtle bob */
      headRef.current.rotation.x = lr(headRef.current.rotation.x, 0.04, 3);
      headRef.current.rotation.y = lr(headRef.current.rotation.y, Math.sin(t * 0.35) * 0.08, 2);

      /* Body Y: footfall rise + breathe */
      rootRef.current.position.set(
        worldPos.current.x,
        worldPos.current.y + Math.abs(Math.sin(phase)) * 0.022 + breathe,
        0,
      );
      rootRef.current.rotation.x = lr(rootRef.current.rotation.x, 0, 3);
      rootRef.current.rotation.z = lr(rootRef.current.rotation.z, 0, 4);

      /* ── REPAIR ────────────────────────────────────────────────────────── */
    } else if (action === 'repair') {
      const CT = 16;
      const ct = t % CT;

      rootRef.current.position.set(worldPos.current.x, worldPos.current.y + breathe, 0);

      /* Phase 0 (0–2.5s): walk up, lean in, examine */
      if (ct < 2.5) {
        torsoRef.current.rotation.x = lr(torsoRef.current.rotation.x, 0.44, 3);
        torsoRef.current.rotation.z = lr(torsoRef.current.rotation.z, 0, 5);

        rArmRef.current.rotation.x = lr(rArmRef.current.rotation.x, -1.05, 3);
        rArmRef.current.rotation.z = lr(rArmRef.current.rotation.z, -0.18, 3);

      rArmRef.current.rotation.x = lr(rArmRef.current.rotation.x, 0.08 + Math.sin(dp * 1.0) * 0.035, 1.8);
      rArmRef.current.rotation.z = lr(rArmRef.current.rotation.z, -(0.52 + Math.sin(dp * 0.8) * 0.030), 1.8);
      lForeRef.current.rotation.x = lr(lForeRef.current.rotation.x, 0.05, 2);
      rForeRef.current.rotation.x = lr(rForeRef.current.rotation.x, 0.05, 2);

      /* Legs loosely together, drift */
      lLegRef.current.rotation.x = lr(lLegRef.current.rotation.x, 0.04 + Math.sin(dp * 1.2) * 0.025, 1.5);
      rLegRef.current.rotation.x = lr(rLegRef.current.rotation.x, 0.04 + Math.sin(dp * 0.9) * 0.025, 1.5);
      lShinRef.current.rotation.x = lr(lShinRef.current.rotation.x, 0, 2);
      rShinRef.current.rotation.x = lr(rShinRef.current.rotation.x, 0, 2);

      torsoRef.current.rotation.x = lr(torsoRef.current.rotation.x, 0, 2);
      torsoRef.current.rotation.z = lr(torsoRef.current.rotation.z, 0, 3);

      /* Head looks around slowly */
      headRef.current.rotation.x = lr(headRef.current.rotation.x, -0.06, 1.5);
      headRef.current.rotation.y = Math.sin(dp * 0.48) * 0.20;
    }

    /* ── Boost shimmy ──────────────────────────────────────────────────── */
    if (boostRef.current > 0) {
      rootRef.current.position.x += Math.sin(t * 22) * boostRef.current * 0.022;
      boostRef.current = Math.max(0, boostRef.current - dt * 2.8);
    } else if (action !== 'fly') {
      rootRef.current.rotation.z = THREE.MathUtils.lerp(rootRef.current.rotation.z, 0, dt * 4);
    }

    /* ── Facing direction ──────────────────────────────────────────────── */
    const spd = Math.hypot(vel.current.x, vel.current.y);
    if (spd > 0.004) {
      rootRef.current.rotation.y = lr(
        rootRef.current.rotation.y,
        vel.current.x > 0 ? Math.PI / 2 : -Math.PI / 2,
        5,
      );
    } else {
      rootRef.current.rotation.y = lr(
        rootRef.current.rotation.y,
        Math.sin(t * 0.22) * 0.10,
        1.5,
      );
    }

    /* ── Scale on hover ────────────────────────────────────────────────── */
    const ts = hovered ? 0.33 : 0.29;
    rootRef.current.scale.setScalar(lr(rootRef.current.scale.x, ts, 10));
  });

  /* ═════════════════════════════════════════════════════════════════════
     JSX  –  Full EMU suit assembly
  ═════════════════════════════════════════════════════════════════════ */
  return (
    <>
      <Tether worldPosRef={worldPos} />

      <group
        ref={rootRef}
        scale={0.29}
        onPointerOver={() => setHovered(true)}
        onPointerOut={() => setHovered(false)}
        onClick={() => { boostRef.current = 1; }}
      >
        {/* ── Dynamic lights ── */}
        <pointLight position={[0, 1.65, 0.38]} color="#3a7fff" intensity={hovered ? 1.5 : 0.75} distance={3.5} decay={2} />
        <pointLight position={[0, -0.85, -0.42]} color="#ff7b00" intensity={flying ? 2.4 : 0.18} distance={2.8} decay={2} />
        <pointLight position={[0.40, 0.22, 0.34]} color="#aaddff" intensity={repairing ? 1.9 : 0} distance={2.2} decay={2} />

        {/* ══ HEAD / HELMET ASSEMBLY ══════════════════════════════════════ */}
        <group ref={headRef} position={[0, 1.10, 0]}>
          {/* Neck ring */}
          <mesh geometry={GEO.neckRing} material={MAT.ring} rotation={[Math.PI / 2, 0, 0]} position={[0, -0.22, 0]} />
          {/* Helmet bubble */}
          <mesh geometry={GEO.helmBubble} material={MAT.helm} />
          {/* Gold visor – positioned on front-lower hemisphere */}
          <mesh
            geometry={GEO.visor}
            material={MAT.visor}
            position={[0, -0.04, 0.07]}
            rotation={[-0.38, 0, 0]}
            scale={[1.03, 1.03, 1.03]}
          />
          {/* American flag patch */}
          <mesh geometry={GEO.badge} material={MAT.patch} position={[0.32, 0.05, 0.14]} rotation={[0, -0.9, 0]} />
        </group>

        {/* ══ TORSO ASSEMBLY ════════════════════════════════════════════ */}
        <group ref={torsoRef} position={[0, 0.26, 0]}>
          {/* HUT – Hard Upper Torso */}
          <mesh geometry={GEO.hut} material={MAT.hut} />
          {/* D-ring tether attach on chest */}
          <mesh geometry={GEO.dRing} material={MAT.dark} position={[0, 0.18, 0.36]} rotation={[Math.PI / 2, 0, 0]} />
          {/* Chest patch / badge */}
          <mesh geometry={GEO.badge} material={MAT.patch} position={[-0.2, 0.14, 0.35]} rotation={[0.15, 0, 0]} />
          {/* LTA – Lower Torso Assembly */}
          <mesh geometry={GEO.lta} material={MAT.suit} position={[0, -0.50, 0]} />
          {/* Hip ring */}
          <mesh geometry={GEO.hipRing} material={MAT.ring} rotation={[Math.PI / 2, 0, 0]} position={[0, -0.28, 0]} />

          {/* ── PLSS BACKPACK ── */}
          <group position={[0, 0.12, -0.32]}>
            <mesh geometry={GEO.plssBox} material={MAT.plss} />
            {/* Oxygen tanks (left + right) */}
            <mesh geometry={GEO.oxyTank} material={MAT.tank} position={[-0.14, 0, -0.09]} />
            <mesh geometry={GEO.oxyTank} material={MAT.tank} position={[0.14, 0, -0.09]} />
            {/* Control panel on bottom-front */}
            <mesh geometry={GEO.panel} material={MAT.panel} position={[0, -0.22, 0.12]} />
          </group>

          {/* ══ LEFT ARM  ════════════════════════════════════════════════ */}
          <group ref={lArmRef} position={[-0.44, 0.16, 0]}>
            {/* Shoulder ring */}
            <mesh geometry={GEO.shoulderRing} material={MAT.ring} rotation={[0, 0, Math.PI / 2]} position={[0, 0.10, 0]} />
            <mesh geometry={GEO.upperArm} material={MAT.suit} position={[0, -0.22, 0]} />
            {/* Elbow – lower arm pivots here */}
            <group ref={lForeRef} position={[0, -0.46, 0]}>
              <mesh geometry={GEO.lowerArm} material={MAT.suit} position={[0, -0.20, 0]} />
              {/* Wrist ring */}
              <mesh geometry={GEO.wristRing} material={MAT.ring} rotation={[0, 0, Math.PI / 2]} position={[0, -0.38, 0]} />
              <mesh geometry={GEO.hand} material={MAT.glove} position={[0, -0.52, 0]} />
            </group>
          </group>

          {/* ══ RIGHT ARM ════════════════════════════════════════════════ */}
          <group ref={rArmRef} position={[0.44, 0.16, 0]}>
            <mesh geometry={GEO.shoulderRing} material={MAT.ring} rotation={[0, 0, Math.PI / 2]} position={[0, 0.10, 0]} />
            <mesh geometry={GEO.upperArm} material={MAT.suit} position={[0, -0.22, 0]} />

        {/* ── Repair props & particles ── */}
        {repairing && <Wrench />}
        <WeldingSparks active={repairing} />
        <ThrusterParticles active={flying} ox={-0.14} oy={-0.88} oz={-0.40} />

        {/* ── Thought bubble ── */}

/* ═══════════════════════════════════════════════════════════════════════════
   DEBRIS FIELD
═══════════════════════════════════════════════════════════════════════════ */
function DebrisField() {
  const items = useMemo(() =>
    Array.from({ length: 16 }, (_, i) => ({
      pos: [(Math.random() - 0.5) * 22, (Math.random() - 0.5) * 16, (Math.random() - 0.5) * 9],
      size: Math.random() * 0.16 + 0.04,
      rspd: [(Math.random() - 0.5) * 0.005, (Math.random() - 0.5) * 0.004, (Math.random() - 0.5) * 0.003],
      col: i % 3 === 0 ? '#e0e0e0' : i % 3 === 1 ? '#1a6fdb' : '#5a5a5a',
    })), []);

  const refs = useRef([]);
  useFrame(() => {
    refs.current.forEach((m, i) => {
      if (!m) return;
      m.rotation.x += items[i].rspd[0];
      m.rotation.y += items[i].rspd[1];
      m.rotation.z += items[i].rspd[2];
    });
  });

  return (
    <>
      {items.map((d, i) => (
          <dodecahedronGeometry args={[d.size, 0]} />
          <meshStandardMaterial color={d.col} metalness={0.44} roughness={0.56} transparent opacity={0.44} />
        </mesh>
      ))}
    </>
  );
}
